// --- 2. the metadata calculator (above 'export const RemotionRoot') ---
const apertureMetadata = ({props}: {props: Record<string, unknown>}) => {
  const d = Number(props?.durationInFrames);
  return Number.isFinite(d) && d > 0 ? {durationInFrames: Math.round(d)} : {};
};
